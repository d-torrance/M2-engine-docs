26const Monoid* degreeMonoid(const std::vector<std::string>& names)
27{
28 std::vector<int> wts;
29 for (int i=0; i<names.size(); i++)
30 wts.push_back(-1);
31 MonomialOrdering* mo = MonomialOrderings::join
32 ({
33 MonomialOrderings::Weights(wts),
34 MonomialOrderings::GroupLex(names.size()),
35 MonomialOrderings::PositionUp()
36 });
37
38 return Monoid::create(mo,
39 IM2_Ring_trivial_polyring()->cast_to_PolynomialRing(),
40 names,
41 {},
42 {});
43}

103const PolynomialRing* simplePolynomialRing(int p, const std::vector<std::string>& names)
104{
105 // if p is 0, use QQ.
106 // degrees are all set to 1. (degree ring has one variable)
107 // heft is 1.
108 // monomial order is grevlex.
109
110 const Ring *kk = (p > 0 ? rawARingZZpFlint(p) : IM2_Ring_QQ());
111 if (kk == nullptr) return nullptr; // one of these routines would have made an error.
112
113 MonomialOrdering* monorder = MonomialOrderings::join
114 ({
115 MonomialOrderings::GRevLex(names.size()),
116 MonomialOrderings::PositionUp()
117 });
118
119 return simplePolynomialRing(kk, names, monorder);
120}

122const WeylAlgebra* simpleWeylAlgebra(long p,
123 const std::vector<std::string> names,
124 const std::vector<int> comms,
125 const std::vector<int> derivs)
126{
127 // if p is 0, use QQ.
128 // degrees are all set to 1. (degree ring has one variable)
129 // heft is 1.
130 // monomial order is grevlex.
131
132 const Ring *kk = (p > 0 ? rawARingZZpFlint(p) : IM2_Ring_QQ());
133 if (kk == nullptr) return nullptr; // one of these routines would have made an error.
134
135 MonomialOrdering* monorder = MonomialOrderings::join
136 ({
137 MonomialOrderings::GRevLex(names.size()),
138 MonomialOrderings::PositionUp()
139 });
140
141 int n = static_cast<int>(comms.size());
142 std::vector<int> degs(2*n, -1);
143 std::fill_n(degs.begin(), n, 1);
144
145 const Monoid* M = Monoid::create(
146 monorder,
147 degreeRing(1),
148 names,
149 degs,
150 {1});
151
152 M2_arrayint derivs1 = stdvector_to_M2_arrayint(derivs);
153 M2_arrayint comms1 = stdvector_to_M2_arrayint(comms);
154 auto W = WeylAlgebra::create(kk,
155 M,
156 derivs1,
157 comms1,
158 -1);
159
160 return W;
161}

164const Matrix* idealFromStrings(const PolynomialRing* R,
165 const std::vector<std::string>& polys)
166{
167 auto varnames = R->getMonoid()->variableNames();
168 BasicPolyList bpList;
169 for (const auto& s : polys)

171 return toMatrix(R->make_FreeModule(1), bpList);

174const Matrix* computeGB(const Matrix* M)
175{
176 M2_arrayint weights = stdvector_to_M2_arrayint(std::vector<int>{});
177 Computation* C = IM2_GB_make(M,
178 false, // collect_syz
179 0, // n_rows_to_keep
180 weights,
181 false, // use_max_degree
182 0, // max_degree
183 0, // algorithm (default)
184 0, // strategy (default)
185 10); // max_reduction_count (engine default)
186 if (C == nullptr) return nullptr;
187 rawStartComputation(C);
188 return rawGBGetMatrix(C);
189}

191const Ring* simpleQuotientRing(const PolynomialRing* R,
192 const std::vector<std::string>& generators)
193{
194 const Matrix* I = idealFromStrings(R, generators);
195 if (I == nullptr) return nullptr;
196 const Matrix* gb = computeGB(I);
197 if (gb == nullptr) return nullptr;
198 return PolynomialRing::create_quotient(R, gb);
199}
