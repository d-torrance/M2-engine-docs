6// A is m x n, row-major. Tests whether {x : Ax >= 0} is full-dimensional.
7ConeResult coneInteriorPoint(int m, int n, const std::vector<int>& A) {
8 glp_prob* lp = glp_create_prob();
9 glp_set_obj_dir(lp, GLP_MAX);
10
11 // Rows: A_i x - t >= 0
12 glp_add_rows(lp, m);
13 for (int i = 1; i <= m; ++i)
14 glp_set_row_bnds(lp, i, GLP_LO, 0.0, 0.0);
15
16 // Cols: x_1..x_n bounded in [-1, 1], then t >= 0
17 glp_add_cols(lp, n + 1);
18 for (int j = 1; j <= n; ++j) {
19 glp_set_col_bnds(lp, j, GLP_DB, -1.0, 1.0);
20 glp_set_obj_coef(lp, j, 0.0);
21 }
22 glp_set_col_bnds(lp, n + 1, GLP_LO, 0.0, 0.0);
23 glp_set_obj_coef(lp, n + 1, 1.0); // maximize t
24
25 // Build sparse matrix in GLPK's 1-indexed triplet form
26 std::vector<int> ia(1, 0), ja(1, 0);
27 std::vector<double> ar(1, 0.0);
28 for (int i = 0; i < m; ++i) {
29 for (int j = 0; j < n; ++j) {
30 int aij = A[i * n + j];
31 if (aij != 0) {
32 ia.push_back(i + 1);
33 ja.push_back(j + 1);
34 ar.push_back(static_cast<double>(aij));
35 }
36 }
37 // -1 coefficient on t
38 ia.push_back(i + 1);
39 ja.push_back(n + 1);
40 ar.push_back(-1.0);
41 }
42 glp_load_matrix(lp, static_cast<int>(ia.size()) - 1,
43 ia.data(), ja.data(), ar.data());
44
45 glp_smcp params;
46 glp_init_smcp(&params);
47 params.msg_lev = GLP_MSG_OFF;
48 glp_simplex(lp, &params);
49
50 ConeResult res;
51 res.tStar = glp_get_col_prim(lp, n + 1);
52
53 const double tol = 1e-8;
54 if (res.tStar > tol) {
55 res.fullDimensional = true;
56 res.interiorPoint.resize(n);
57 for (int j = 0; j < n; ++j)
58 res.interiorPoint[j] = glp_get_col_prim(lp, j + 1);
59 } else {
60 res.fullDimensional = false;
61 res.dualCert.resize(m);
62 for (int i = 0; i < m; ++i)
63 // GLPK row duals for >= constraints in a max problem may come
64 // out negative depending on convention; take |.| so y >= 0.
65 res.dualCert[i] = std::abs(glp_get_row_dual(lp, i + 1));
66 }
67 glp_delete_prob(lp);
68 return res;
69}
