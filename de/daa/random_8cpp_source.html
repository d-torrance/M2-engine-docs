44int32_t rawRandomInt(int32_t max)
45/* generate a random number in the range 0..max-1 */
46{
47 if (max <= 0) return 0;
48 int32_t k = RandomSeed / IQ;
49 RandomSeed =
50 IA * (RandomSeed - k * IQ) - IR * k; /* Schrage algorithm to compute
51 idum = (IA*idum) mod IM */
52 if (RandomSeed < 0) RandomSeed += IM;
53 return RandomSeed % max;
54}

73void rawSetFareyApproximation(mpq_ptr result, gmp_RR x, gmp_ZZ height)
74/* sets result = the nearest rational to x w/ denominator <= height */
75/* see https://en.wikipedia.org/wiki/Farey_sequence */
76{
77 int sgn;
78 mpfr_t fracpart, tmp1, tmp2;
79 mpz_t intpart, a, b, c, d, q, p;
80 mpq_t tmp3;
81
82 // get integer and fractional parts of |x|
83 sgn = mpfr_sgn(x);
84 mpfr_init2(fracpart, mpfr_get_prec(x));
85 mpfr_abs(fracpart, x, MPFR_RNDN);
86 mpz_init(intpart);
87 mpfr_get_z(intpart, fracpart, MPFR_RNDD);
88 mpfr_frac(fracpart, fracpart, MPFR_RNDN);
89
90 // goal: find nearest rational number to fracpart
91 // start w/ a/b = 0, p/q = 1/2, c/d = 1
92 mpfr_init2(tmp1, mpfr_get_prec(x));
93 mpfr_init2(tmp2, mpfr_get_prec(x));
94 mpz_init_set_ui(a, 0);
95 mpz_init_set_ui(b, 1);
96 mpz_init_set_ui(c, 1);
97 mpz_init_set_ui(d, 1);
98 mpz_init_set_ui(p, 1);
99 mpz_init_set_ui(q, 2);
100 mpq_init(tmp3);
101
102 // compute mediant p/q = (a+c)/(b+d) until q > height
103 while (mpz_cmp(q, height) <= 0) {
104 mpfr_mul_z(tmp1, fracpart, q, MPFR_RNDN);
105 // check if fracpart is to the left or right of p/q
106 // and update a/b or c/d accordingly
107 if (mpfr_cmp_z(tmp1, p) <= 0) {
108 mpz_set(c, p);
109 mpz_set(d, q);
110 } else {
111 mpz_set(a, p);
112 mpz_set(b, q);
113 }
114 mpz_add(p, a, c);
115 mpz_add(q, b, d);
116 }
117
118 // now check which endpoint is closest
119 // tmp1 = fracpart - a/b
120 mpfr_set_z(tmp1, a, MPFR_RNDN);
121 mpfr_neg(tmp1, tmp1, MPFR_RNDN);
122 mpfr_div_z(tmp1, tmp1, b, MPFR_RNDN);
123 mpfr_add(tmp1, tmp1, fracpart, MPFR_RNDN);
124
125 // tmp2 = c/d - fracpart
126 mpfr_set_z(tmp2, c, MPFR_RNDN);
127 mpfr_div_z(tmp2, tmp2, d, MPFR_RNDN);
128 mpfr_sub(tmp2, tmp2, fracpart, MPFR_RNDN);
129
130 if (mpfr_cmp(tmp1, tmp2) <= 0) {
131 mpq_set_z(result, a);
132 mpq_set_z(tmp3, b);
133 } else {
134 mpq_set_z(result, c);
135 mpq_set_z(tmp3, d);
136 }
137
138 // finally, add back intpart and negate if x < 0
139 mpq_div(result, result, tmp3);
140 mpq_set_z(tmp3, intpart);
141 mpq_add(result, result, tmp3);
142 mpq_set_si(tmp3, sgn, 1);
143 mpq_mul(result, result, tmp3);
144 mpq_canonicalize(result);
145
146 mpz_clears(intpart, a, b, c, d, p, q, nullptr);
147 mpfr_clears(fracpart, tmp1, tmp2, nullptr);
148 mpq_clear(tmp3);
149}

160void rawSetRandomQQ(mpq_ptr result, gmp_ZZ height)
161/* returns random a/b, where 1 <= b <= height, 1 <= a <= height */
162/* if height is the null pointer, use the default height */
163{
164 mpz_t d;
165
166 mpz_init(d);
167 if (height == nullptr) height = maxHeight;
168 if (mpz_cmp_si(height, 0) <= 0)
169 throw exc::engine_error("expected a positive height");
170
171 while (true) {
172 mpz_urandomm(mpq_numref(result), state, height);
173 mpz_urandomm(mpq_denref(result), state, height);
174 mpz_add_ui(mpq_numref(result), mpq_numref(result), 1);
175 mpz_add_ui(mpq_denref(result), mpq_denref(result), 1);
176 mpz_gcd(d, mpq_numref(result), mpq_denref(result));
177 if (mpz_cmp_ui(d, 1) == 0)
178 break;
179 }
180
181 mpz_clear(d);
182}
