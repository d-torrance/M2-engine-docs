47// This struct is needed to create sets of vectors
48struct VectorHash {
49 size_t operator()(const std::vector<int>& v) const {
50 size_t seed = v.size();
51 size_t x;
52 for(auto c : v) {
53 x = c;
54 x = ((x >> 16) ^ x) + 0xdf3b45d9f3b;
55 x = ((x << 16) ^ x) + 0xdf3b45d9f3b;
56 seed ^= x + 0x779b99e3779b9 + (seed << 12) + (seed >> 4);
57 }
58 return seed;
59 }
60};

95// Compute factorials with multiple precision
96MPFloat mpfactorial(int z) {
97 MPFloat res = z;
98 if (z < 0) {
99 throw std::invalid_argument("Input must be non-negative.");
100 }
101 else if (z == 0) {
102 res = 1;
103 }
104 for (int i = z-1; i > 1; i--){
105 res *= i;
106 }
107 return res;
108}

110// Compute digamma for integers with multiple precision
111MPFloat mpdigamma(int z, const MPFloat &neg_em) {
112 if (z <= 0) {
113 throw std::invalid_argument("Digamma is infinite or complex.");
114 }
115 MPFloat res = neg_em;
116 MPFloat tmpVar;
117 for (int i = 1; i < z; i++){
118 tmpVar = i;
119 res += 1/tmpVar;
120 }
121 return res;
122}

124// Compute trigamma for integers with multiple precision
125MPFloat mptrigamma(int z, const MPFloat &pi2d6) {
126 if (z <= 0) {
127 throw std::invalid_argument("Trigamma is infinite or complex.");
128 }
129 MPFloat res = pi2d6;
130 MPFloat tmpVar;
131 for (int i = 1; i < z; i++){
132 tmpVar = i;
133 res -= 1/(tmpVar*tmpVar);
134 }
135 return res;
136}

138// Parse a vector from std::cin
139std::vector<int> ReadVector(std::istream &ist) {
140 char c;
141 int i;
142 std::vector<int> vec;
143 ist >> std::ws >> c;
144 if (c == '(' || c == '[' || c == '{') {
145 while (ist >> std::ws >> c) {
146 if (c == ')' || c == ']' || c == '}') {
147 break;
148 }
149 else if (isspace(c)) {
150 continue;
151 }
152 else if (c == ',') {
153 continue;
154 }
155 else if(isdigit(c) || c == '-') {
156 ist.putback(c);
157 ist >> i;
158 vec.push_back(i);
159 }
160 else {
161 throw std::invalid_argument("Failed to parse input as integer.");
162 }
163 }
164 }
165 else {
166 throw std::invalid_argument("Missing vector opening bracket.");
167 }
168 ist.clear(std::ios::goodbit);
169 return vec;
170}

172// Parse a vector of vectors from std::cin
173std::vector<std::vector<int> > ReadVectors(std::istream &ist,
174 bool allow_diff_lengths=false) {
175 char c;
176 unsigned int vec_size = 0;
177 std::vector<std::vector<int> > vecs;
178 std::vector<int> tmp_vec;
179 ist >> std::ws >> c;
180 if (c == '(' || c == '[' || c == '{') {
181 while (ist >> std::ws >> c) {
182 if (c == ')' || c == ']' || c == '}') {
183 break;
184 }
185 else if (isspace(c)) {
186 continue;
187 }
188 else if (c == ',') {
189 continue;
190 }
191 else if (c == '(' || c == '[' || c == '{') {
192 ist.putback(c);
193 tmp_vec = ReadVector(ist);
194 if (!tmp_vec.size()){
195 throw std::invalid_argument("Read vector with zero size.");
196 }
197 if (!vec_size) {
198 vec_size = tmp_vec.size();
199 }
200 else if (!allow_diff_lengths && vec_size != tmp_vec.size()){
201 throw std::invalid_argument("All vectors must be the same size.");
202 }
203 vecs.push_back(tmp_vec);
204 }
205 else {
206 throw std::invalid_argument("Missing vector opening bracket.");
207 }
208 }
209 }
210 else{
211 throw std::invalid_argument("Missing matrix opening bracket.");
212 }
213 ist.clear(std::ios::goodbit);
214 return vecs;

217// Completes a given list of curves by making sure that all possible sums are
218// in the list up to the specified maximum degree.
219void CheckCompletenessThr(VectorSet &vset, const IntMatrix &check_vecs,
220 IntMatrix &new_vecs,
221 const IntMatrix &hilbert_basis,
222 IntVector &grading_vec, int max_deg,
223 int &curr_task, int n_curves, std::mutex &m,
224 std::mutex &task_mut, int min_mem) {
225 int h11 = check_vecs[0].size();
226 int tmp_deg;
227 std::vector<int> tmp_vec(h11);
228 while (true) {
229 if (GetAvailableSystemMemory() < min_mem) {
230 throw std::runtime_error("Memory is running low. "
231 "Exitting to prevent crash...");
232 }
233 task_mut.lock();
234 int i = curr_task;
235 curr_task++;
236 task_mut.unlock();
237 if (i >= n_curves) {
238 break;
239 }
240 for (unsigned int j = 0; j < hilbert_basis.size(); j++) {
241 tmp_deg = 0;
242 for (int k = 0; k < h11; k++) {
243 tmp_vec[k] = check_vecs[i][k] + hilbert_basis[j][k];
244 tmp_deg += tmp_vec[k] * grading_vec[k];
245 }
246 // Discard if the vector exceeds the maximum degree
247 if (tmp_deg > max_deg) {
248 continue;
249 }
250 // Search in the set and add if missing
251 m.lock();
252 auto search = vset.find(tmp_vec);
253 if (search != vset.end()){
254 m.unlock();
255 continue;
256 }
257 vset.insert(tmp_vec);
258 new_vecs.push_back(tmp_vec);
259 m.unlock();
260 }
261 }
262}

264// Removes curves that are not in the Hilbert basis. This is done by checking
265// that curves are not sums of up to 5 other curves.
266void RemoveNonGensThr(const VectorSet &vset, const IntMatrix &vecs,
267 IntMatrix &vecs_to_remove, const IntVector &grading_vec,
268 int max_deg, int &curr_task, int n_curves, std::mutex &m,
269 std::mutex &task_mut) {
270 int h11 = vecs[0].size();
271 int tmp_deg;
272 std::vector<int> tmp_vec(h11);
273 while (true) {
274 task_mut.lock();
275 int i = curr_task;
276 curr_task++;
277 task_mut.unlock();
278 if (i >= n_curves) {
279 break;
280 }
281 for (int j = i; j < n_curves; j++) {
282 tmp_deg = 0;
283 for (int k = 0; k < h11; k++) {
284 tmp_vec[k] = vecs[i][k] + vecs[j][k];
285 tmp_deg += tmp_vec[k] * grading_vec[k];
286 }
287 // Discard if the vector exceeds the maximum degree
288 if (tmp_deg > max_deg) {
289 continue;
290 }
291 // Search in the set and erase if found
292 auto search2 = vset.find(tmp_vec);
293 if (search2 != vset.end()) {
294 m.lock();
295 vecs_to_remove.push_back(tmp_vec);
296 m.unlock();
297 }
298 for (unsigned int k = j; k < vecs.size(); k++) {
299 tmp_deg = 0;
300 for (int kk = 0; kk < h11; kk++) {
301 tmp_vec[kk] = vecs[i][kk] + vecs[j][kk] + vecs[k][kk];
302 tmp_deg += tmp_vec[kk] * grading_vec[kk];
303 }
304 // Discard if the vector exceeds the maximum degree
305 if (tmp_deg > max_deg) {
306 continue;
307 }
308 // Search in the set and erase if found
309 auto search3 = vset.find(tmp_vec);
310 if (search3 != vset.end()) {
311 m.lock();
312 vecs_to_remove.push_back(tmp_vec);
313 m.unlock();
314 }
315 for (unsigned int l = k; l < vecs.size(); l++) {
316 tmp_deg = 0;
317 for (int kk = 0; kk < h11; kk++) {
318 tmp_vec[kk] = (vecs[i][kk] + vecs[j][kk] + vecs[k][kk]
319 + vecs[l][kk]);
320 tmp_deg += tmp_vec[kk] * grading_vec[kk];
321 }
322 // Discard if the vector exceeds the maximum degree
323 if (tmp_deg > max_deg) {
324 continue;
325 }
326 // Search in the set and erase if found
327 auto search4 = vset.find(tmp_vec);
328 if (search4 != vset.end()) {
329 m.lock();
330 vecs_to_remove.push_back(tmp_vec);
331 m.unlock();
332 }
333 for (unsigned int ll = l; ll < vecs.size(); ll++) {
334 tmp_deg = 0;
335 for (int kk = 0; kk < h11; kk++) {
336 tmp_vec[kk] = (vecs[i][kk] + vecs[j][kk] + vecs[k][kk]
337 + vecs[l][kk] + vecs[ll][kk]);
338 tmp_deg += tmp_vec[kk] * grading_vec[kk];
339 }
340 // Discard if the vector exceeds the maximum degree
341 if (tmp_deg > max_deg) {
342 continue;
343 }
344 // Search in the set and erase if found
345 auto search5 = vset.find(tmp_vec);
346 if (search5 != vset.end()) {
347 m.lock();
348 vecs_to_remove.push_back(tmp_vec);
349 m.unlock();
350 }
351 }
352 }
353 }
354 }
355 }

358// Fills out the relevant part of the past light cone
359void FillPastLightConeThr(VectorSet &vset, const IntMatrix &check_vecs,
360 const VectorSet &pset, IntMatrix &new_vecs,
361 const IntMatrix &hilbert_basis,
362 IntVector &grading_vec, int max_deg, int &curr_task,
363 int n_curves, std::mutex &m, std::mutex &task_mut,
364 int min_mem) {
365 int h11 = check_vecs[0].size();
366 int tmp_deg;
367 std::vector<int> tmp_vec(h11);
368 while (true) {
369 if (GetAvailableSystemMemory() < min_mem) {
370 throw std::runtime_error("Memory is running low. "
371 "Exitting to prevent crash...");
372 }
373 task_mut.lock();
374 int i = curr_task;
375 curr_task++;
376 task_mut.unlock();
377 if (i >= n_curves) {
378 break;
379 }
380 for (unsigned int j = 0; j < hilbert_basis.size(); j++) {
381 tmp_deg = 0;
382 for (int k = 0; k < h11; k++) {
383 tmp_vec[k] = check_vecs[i][k] + hilbert_basis[j][k];
384 tmp_deg += tmp_vec[k] * grading_vec[k];
385 }
386 // Discard if the vector exceeds the maximum degree
387 if (tmp_deg > 2*max_deg) {
388 continue;
389 }
390 // Discard if the vector is not in the past light cone
391 auto search = pset.find(tmp_vec);
392 if (search == pset.end()) {
393 continue;
394 }
395 // Search in the set and add if missing
396 m.lock();
397 search = vset.find(tmp_vec);
398 if (search != vset.end()){
399 m.unlock();
400 continue;
401 }
402 vset.insert(tmp_vec);
403 new_vecs.push_back(tmp_vec);
404 m.unlock();
405 }
406 }

409// Computes half of the past light cone. I.e. it takes the cone that was
410// computed and flips it and sets the apex to be the relevant curve.
411void HalfPastLightConeThr(VectorSet &vset, VectorSet &f_vset,
412 std::vector<int> &vec, std::mutex &m) {
413 int h11 = vec.size();
414 std::vector<int> tmp_vec(h11);
415 for (auto &vec2 : vset) {
416 for (int i = 0; i < h11; i++) {
417 tmp_vec[i] = vec[i] - vec2[i];
418 }
419 auto search = f_vset.find(tmp_vec);
420 if (search != vset.end()){
421 continue;
422 }
423 m.lock();
424 f_vset.insert(tmp_vec);
425 m.unlock();
426 }
427}

429// Trims the remaining curves that are not in the past light cone
430void TrimPastLightConeThr(VectorSet &vset, VectorSet &f_vset,
431 std::vector<int> &vec, std::mutex &m) {
432 int h11 = vec.size();
433 std::vector<int> tmp_vec(h11);
434 for (auto &vec2 : vset) {
435 for (int i = 0; i < h11; i++) {
436 tmp_vec[i] = vec[i] - vec2[i];

440 if (search == vset.end()){

444 if (search != vset.end()){
445 continue;
446 }
447 m.lock();
448 f_vset.insert(tmp_vec);
449 m.unlock();
450 }

480// Computes the product of two polynomials
481Polynomial PolyProd(const Polynomial &p1, const Polynomial &p2,
482 const IntMatrix &curves, const IntVector &degs,
483 const VecToIntDict &curve_dict) {
484 Polynomial res;
485 int d1, d2, ind;
486 int h11 = curves[0].size();
487 int max_deg = degs[degs.size()-1];
488 IntVector tmp_vec_h11(h11);
489 VecToIntDict::const_iterator search;
490 std::unordered_map<int,MPFloat>::const_iterator search2;
491 const Polynomial *pshort;
492 const Polynomial *plong;
493 if (p1.nonzero.size() < p2.nonzero.size()) {
494 pshort = &p1;
495 plong = &p2;
496 }
497 else {
498 pshort = &p2;
499 plong = &p1;
500 }
501 for (auto &i : pshort->nonzero) {
502 d1 = degs[i];
503 for (auto &j : plong->nonzero) {
504 d2 = degs[j];
505 if (d1+d2 > max_deg) {
506 break;
507 }
508 for (int k = 0; k < h11; k++) {
509 tmp_vec_h11[k] = curves[i][k] + curves[j][k];
510 }
511 search = curve_dict.find(tmp_vec_h11);
512 if (search == curve_dict.end()) {
513 continue;
514 }
515 ind = search->second;
516 search2 = res.coeffs.find(ind);
517 if (search2 == res.coeffs.end()) {
518 res.coeffs[ind] = pshort->coeffs.at(i) * plong->coeffs.at(j);
519 res.nonzero.push_back(ind);
520 }
521 else {
522 res.coeffs[ind] += pshort->coeffs.at(i) * plong->coeffs.at(j);
523 }
524 }
525 }
526 std::sort(res.nonzero.begin(), res.nonzero.end());
527 return res;
528}

576// Computes the sum of polynomials in place
577void PolySumIP(Polynomial &p1, const Polynomial &p2) {
578 bool resort = false;
579 std::unordered_map<int,MPFloat>::const_iterator search;
580 for (auto &c : p2.coeffs) {
581 search = p1.coeffs.find(c.first);
582 if (search == p1.coeffs.end()) {
583 p1.coeffs[c.first] = c.second;
584 p1.nonzero.push_back(c.first);
585 resort = true;
586 }
587 else {
588 p1.coeffs[c.first] += c.second;
589 }
590 }
591 if (resort) {
592 std::sort(p1.nonzero.begin(), p1.nonzero.end());
593 }
594}

674// Computes the inverse of a polynomial
675Polynomial PolyInv(const Polynomial &p, const IntMatrix &curves,
676 const IntVector &degs, const VecToIntDict &curve_dict) {
677 Polynomial res;
678 res.nonzero.push_back(0);
679 res.coeffs[0] = 1;
680 if (p.nonzero[0] != 0 || !p.coeffs.at(0)) {
681 throw std::invalid_argument("Polynomial is not invertible.");
682 }
683 int max_deg = degs[degs.size()-1];
684 Polynomial p0 = p;
685 p0.nonzero.erase(p0.nonzero.begin());
686 p0.coeffs.erase(0);
687 PolyDivIP(p0, p.coeffs.at(0));
688 Polynomial tmp_poly = res;
689 int min_deg = PolyMinDeg(p0, degs);
690 for (int i = 1; i < 1+max_deg/min_deg; i++) {
691 tmp_poly = PolyProd(tmp_poly, p0, curves, degs, curve_dict);
692 if (i&1) {
693 PolySumFactIP(res, tmp_poly, -1);
694 }
695 else {
696 PolySumIP(res, tmp_poly);
697 }
698 }
699 PolyDivIP(res, p.coeffs.at(0));
700 return res;
701}

703// Computes the nonnegative power of a polynomial
704Polynomial PolyPow(const Polynomial &p, unsigned int n,
705 const IntMatrix &curves, const IntVector &degs,
706 const VecToIntDict &curve_dict) {
707 Polynomial res;
708 res.nonzero.push_back(0);
709 res.coeffs[0] = 1;
710 if (n == 0) {
711 return res;
712 }
713 else if (n == 1) {
714 res = p;
715 return res;
716 }
717 int max_deg = degs[degs.size()-1];
718 Polynomial tmp_poly = PolyTrunc(p, max_deg-(n-1)*PolyMinDeg(p, degs), degs);
719 while (true) {
720 if (n & 1) {
721 res = PolyProd(res, tmp_poly, curves, degs, curve_dict);
722 }
723 n >>= 1;
724 if (!n) {
725 break;
726 }
727 tmp_poly = PolyProd(tmp_poly, tmp_poly, curves, degs, curve_dict);
728 }
729 return res;
730}

732// Computes the exponential of a polynomial (and its negative)
733void PolyExp(Polynomial &res_pos, Polynomial &res_neg, const Polynomial &p,
734 const IntMatrix &curves, const IntVector &degs,
735 const VecToIntDict &curve_dict) {
736 bool has_const = (p.nonzero.size() > 0 && p.nonzero[0] == 0);
737 res_pos = Polynomial();
738 res_pos.nonzero.push_back(0);
739 res_pos.coeffs[0] = 1;
740 res_neg = res_pos;
741 Polynomial p0 = p;
742 if (has_const) {
743 p0.nonzero.erase(p0.nonzero.begin());
744 p0.coeffs.erase(0);
745 }
746 Polynomial tmp_poly = res_pos;
747 int min_deg = PolyMinDeg(p0, degs);
748 int max_deg = degs[degs.size()-1];
749 MPFloat tmp_var;
750 for (int i = 1; i < 1+max_deg/min_deg; i++) {
751 tmp_poly = PolyProd(tmp_poly, p0, curves, degs, curve_dict);
752 tmp_var = mpfactorial(i);
753 PolySumDivIP(res_pos, tmp_poly, tmp_var);
754 if (i&1) {
755 tmp_var *= -1;
756 }
757 PolySumDivIP(res_neg, tmp_poly, tmp_var);
758 }
759 if (has_const) {
760 tmp_var = exp(p.coeffs.at(0));
761 PolyProdScalarIP(res_pos, tmp_var);
762 PolyProdScalarIP(res_neg, tmp_var);
763 }
764}

766// Computes the dilogarithm of a polynomial
767Polynomial PolyLi2(const Polynomial &p, const IntMatrix &curves,
768 const IntVector &degs, const VecToIntDict &curve_dict) {
769 if ((p.nonzero.size() > 0 && p.nonzero[0] == 0)
770 || p.coeffs.find(0) != p.coeffs.end()) {
771 PolyPrint(p);
772 throw std::invalid_argument("This type of polynomials are not "
773 "supported.");
774 }
775 Polynomial res = p;
776 Polynomial tmp_poly = p;
777 int min_deg = PolyMinDeg(p, degs);
778 int max_deg = degs[degs.size()-1];
779 for (int i = 2; i < 1+max_deg/min_deg; i++) {
780 tmp_poly = PolyProd(tmp_poly, p, curves, degs, curve_dict);
781 PolySumDivIP(res, tmp_poly, i*i);
782 }
783 return res;
784}

786// Removes coefficients with a very small magnitude
787void PolyCleanUp(Polynomial &p, MPFloat &cutoff, bool force_resort=false) {
788 std::vector<int> new_nonzero;
789 for (auto &i : p.nonzero) {
790 if (abs(p.coeffs[i]) < cutoff) {
791 p.coeffs.erase(i);
792 }
793 else {
794 new_nonzero.push_back(i);
795 }
796 }
797 if (force_resort) {
798 std::sort(new_nonzero.begin(), new_nonzero.end());
799 }
800 p.nonzero = new_nonzero;
801}

803// Computes c for curves with zero negative entries in the dot product with Q
804void computeC_0neg(Polynomial &c0, std::vector<Polynomial> &c1,
805 VecToPolyDict &c2, const std::vector<int> &neg0,
806 const IntMatrix &Q, const IntMatrix &Q0s,
807 const IntMatrix &curves_dot_Q,
808 const IntMatrix &curves_dot_Q0s, int prec,
809 std::mutex &c0mutex, std::mutex &c1mutex,
810 std::mutex &c2mutex, int h11, int h11pd, int min_mem,
811 int &curr_task, std::mutex &task_mut, VectorSet &beta_pairs,
812 const MPFloat &neg_em, const MPFloat &pi2d6) {
813 mpfr::mpreal::set_default_prec(prec);
814 std::vector<MPFloat> A(h11);
815 MPFloat tmp_num, c0fact, tmp_final;
816 int i, i_ind, a, b;
817 while (true) {
818 task_mut.lock();
819 i_ind = curr_task;
820 curr_task++;
821 task_mut.unlock();
822 if (i_ind >= neg0.size()) {
823 break;
824 }
825 i = neg0[i_ind];
826 if (GetAvailableSystemMemory() < min_mem) {
827 throw std::runtime_error("Memory is running low. "
828 "Exitting to prevent crash...");
829 }
830 c0fact = mpfactorial(curves_dot_Q0s[0][i]);
831 for (int j = 1; j < Q0s.size(); j++){
832 c0fact *= mpfactorial(curves_dot_Q0s[j][i]);
833 }
834 for (int j = 0; j < h11pd; j++) {
835 c0fact /= mpfactorial(curves_dot_Q[i][j]);
836 }
837 c0mutex.lock();
838 c0.coeffs[i] = c0fact;
839 c0.nonzero.push_back(i);
840 c0mutex.unlock();
841 for (int a = 0; a < h11; a++) {
842 A[a] = Q0s[0][a]*mpdigamma(curves_dot_Q0s[0][i]+1, neg_em);
843 for (int j = 1; j < Q0s.size(); j++){
844 A[a] += Q0s[j][a]*mpdigamma(curves_dot_Q0s[j][i]+1, neg_em);
845 }
846 for (int j = 0; j < h11pd; j++) {
847 A[a] -= Q[a][j]*mpdigamma(curves_dot_Q[i][j]+1, neg_em);
848 }
849 tmp_final = c0fact*A[a];
850 c1mutex.lock();
851 c1[a].coeffs[i] = tmp_final;
852 c1[a].nonzero.push_back(i);
853 c1mutex.unlock();
854 }
855 for (auto &pp : beta_pairs) {
856 a = pp[0];
857 b = pp[1];
858 tmp_num = Q0s[0][a]*Q0s[0][b]*mptrigamma(curves_dot_Q0s[0][i]+1, pi2d6);
859 for (int j = 1; j < Q0s.size(); j++){
860 tmp_num += (Q0s[j][a]*Q0s[j][b]
861 *mptrigamma(curves_dot_Q0s[j][i]+1, pi2d6));
862 }
863 for (int j = 0; j < h11pd; j++) {
864 tmp_num -= Q[a][j]*Q[b][j]*mptrigamma(curves_dot_Q[i][j]+1, pi2d6);
865 }
866 tmp_num += A[a]*A[b];
867 tmp_final = c0fact*tmp_num;
868 c2mutex.lock();
869 c2[{a,b}].coeffs[i] = tmp_final;
870 c2[{a,b}].nonzero.push_back(i);
871 c2mutex.unlock();
872 }
873 }
874 mpfr_free_cache();
875}

877// Computes c for curves with one negative entry in the dot product with Q
878void computeC_1neg(std::vector<Polynomial> &c1, VecToPolyDict &c2,
879 const std::vector<std::tuple<int,int> > &neg1,
880 const IntMatrix &Q, const IntMatrix &Q0s,
881 const IntMatrix &curves_dot_Q,
882 const IntMatrix &curves_dot_Q0s, int prec,
883 std::mutex &c1mutex, std::mutex &c2mutex, int h11,
884 int h11pd, int min_mem, int &curr_task,
885 std::mutex &task_mut, VectorSet &beta_pairs,
886 const MPFloat &neg_em) {
887 mpfr::mpreal::set_default_prec(prec);
888 std::vector<MPFloat> A(h11);
889 MPFloat tmp_fact, tmp_final;
890 int i, k, m, sn;
891 int i_ind, a, b;
892 while (true) {
893 task_mut.lock();
894 i_ind = curr_task;
895 curr_task++;
896 task_mut.unlock();
897 if (i_ind >= neg1.size()) {
898 break;
899 }
900 if (GetAvailableSystemMemory() < min_mem) {
901 throw std::runtime_error("Memory is running low. "
902 "Exitting to prevent crash...");
903 }
904 i = std::get<0>(neg1[i_ind]);
905 k = std::get<1>(neg1[i_ind]);
906 m = curves_dot_Q[i][k];
907 sn = (m&1 ? 1 : -1);
908 tmp_fact = mpfactorial(curves_dot_Q0s[0][i]);
909 for (int j = 1; j < Q0s.size(); j++){
910 tmp_fact *= mpfactorial(curves_dot_Q0s[j][i]);
911 }
912 tmp_fact *= mpfactorial(-m-1);
913 for (int j = 0; j < h11pd; j++) {
914 if (j == k) {
915 continue;
916 }
917 tmp_fact /= mpfactorial(curves_dot_Q[i][j]);
918 }
919 for (int a = 0; a < h11; a++) {
920 A[a] = Q0s[0][a]*mpdigamma(curves_dot_Q0s[0][i]+1, neg_em);
921 for (int j = 1; j < Q0s.size(); j++){
922 A[a] += Q0s[j][a]*mpdigamma(curves_dot_Q0s[j][i]+1, neg_em);
923 }
924 for (int j = 0; j < h11pd; j++) {
925 A[a] -= Q[a][j]*mpdigamma((j==k ? -m : curves_dot_Q[i][j]+1), neg_em);
926 }
927 tmp_final = (sn*Q[a][k])*tmp_fact;
928 c1mutex.lock();
929 c1[a].coeffs[i] = tmp_final;
930 c1[a].nonzero.push_back(i);
931 c1mutex.unlock();
932 }
933 for (auto &pp : beta_pairs) {
934 a = pp[0];
935 b = pp[1];
936 tmp_final = (sn*(Q[a][k]*A[b]+Q[b][k]*A[a]))*tmp_fact;
937 c2mutex.lock();
938 c2[{a,b}].coeffs[i] = tmp_final;
939 c2[{a,b}].nonzero.push_back(i);
940 c2mutex.unlock();
941 }
942 }
943 mpfr_free_cache();
944}

946// Computes c for curves with two negative entries in the dot product with Q
947void computeC_2neg(VecToPolyDict &c2,
948 const std::vector<std::tuple<int,int,int> > &neg2,
949 const IntMatrix &Q, const IntMatrix &Q0s,
950 const IntMatrix &curves_dot_Q,
951 const IntMatrix &curves_dot_Q0s, int prec,
952 std::mutex &c2mutex, int h11, int h11pd, int min_mem,
953 int &curr_task, std::mutex &task_mut,
954 VectorSet &beta_pairs) {
955 mpfr::mpreal::set_default_prec(prec);
956 std::vector<MPFloat> A(h11);
957 MPFloat c0fact, tmp_num;
958 int i, k0, k1, m0, m1;
959 int i_ind, a, b;
960 while (true) {
961 task_mut.lock();
962 i_ind = curr_task;
963 curr_task++;
964 task_mut.unlock();
965 if (i_ind >= neg2.size()) {
966 break;
967 }
968 if (GetAvailableSystemMemory() < min_mem) {
969 throw std::runtime_error("Memory is running low. "
970 "Exitting to prevent crash...");
971 }
972 i = std::get<0>(neg2[i_ind]);
973 k0 = std::get<1>(neg2[i_ind]);
974 k1 = std::get<2>(neg2[i_ind]);
975 m0 = curves_dot_Q[i][k0];
976 m1 = curves_dot_Q[i][k1];
977 c0fact = ((m0+m1)&1 ? -1 : 1);
978 c0fact *= mpfactorial(curves_dot_Q0s[0][i]);
979 for (int j = 1; j < Q0s.size(); j++){
980 c0fact *= mpfactorial(curves_dot_Q0s[j][i]);
981 }
982 c0fact *= mpfactorial(-m0-1);
983 c0fact *= mpfactorial(-m1-1);
984 for (int j = 0; j < h11pd; j++) {
985 if (j == k0 || j== k1) {
986 continue;
987 }
988 c0fact /= mpfactorial(curves_dot_Q[i][j]);
989 }
990 for (auto &pp : beta_pairs) {
991 a = pp[0];
992 b = pp[1];
993 tmp_num = c0fact*(Q[a][k0]*Q[b][k1]+Q[a][k1]*Q[b][k0]);
994 c2mutex.lock();
995 c2[{a,b}].coeffs[i] = tmp_num;
996 c2[{a,b}].nonzero.push_back(i);
997 c2mutex.unlock();
998 }
999 }
1000 mpfr_free_cache();
1001}

1217Polynomial ComputeqN(const Polynomial &closest_curve,
1218 const IntVector &closest_curve_diff,
1219 const std::vector<Polynomial> &expalpha_pos,
1220 const std::vector<Polynomial> &expalpha_neg,

1241// Computes qN and Li2(qN)
1242void ComputeLi2qNThr(const std::vector<int> &qN_to_compute,
1243 IntToPolyDict &computed_qN, IntToPolyDict &computed_Li2qN,
1244 const std::vector<IntToPolyDict> &previous_qN,
1245 const std::vector<IntVector> &previous_qN_ind,
1246 const std::vector<Polynomial> &expalpha_pos,
1247 const std::vector<Polynomial> &expalpha_neg,
1248 const IntMatrix &curves, const IntVector &degs,
1249 const VecToIntDict &curve_dict, int prec, int &curr_task,
1250 std::mutex &task_mut, std::mutex &m, bool computeGW,
1251 int min_mem) {
1252 mpfr::mpreal::set_default_prec(prec);
1253 int h11 = curves[0].size();
1254 Polynomial closest_curve, tmp_qN, tmp_Li2qN, tmp_poly;
1255 IntVector closest_curve_diff(h11), tmp_curve_diff(h11);
1256 double closest_dist, tmp_dist;
1257 VecToIntDict::const_iterator search;
1258 int i, i_ind;
1259 while (true) {
1260 task_mut.lock();
1261 i_ind = curr_task;
1262 curr_task++;
1263 task_mut.unlock();
1264 if (i_ind >= qN_to_compute.size()) {
1265 break;
1266 }
1267 if (GetAvailableSystemMemory() < min_mem) {
1268 throw std::runtime_error("Memory is running low. "
1269 "Exitting to prevent crash...");
1270 }
1271 i = qN_to_compute[i_ind];
1272 closest_curve = Polynomial();
1273 closest_curve.nonzero.push_back(i);
1274 closest_curve.coeffs[i] = 1;
1275 closest_curve_diff = curves[i];
1276 closest_dist = 0;
1277 for (int j = 0; j < h11; j++) {
1278 if (closest_curve_diff[j]) {
1279 closest_dist += std::log2(std::abs(closest_curve_diff[j]))+1;
1280 }
1281 }
1282 // Now we check to see if there is a better starting curve
1283 for (int j = 0; j < previous_qN_ind.size(); j++) {
1284 for (auto k : previous_qN_ind[j]) {
