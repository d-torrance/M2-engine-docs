8// Convert a BasicPoly (parsed polynomial with mpz_class coefficients and
9// varpower monomials) into a ring_elem in the given PolynomialRing.
10static ring_elem basicPolyToRingElem(const PolynomialRing *P, const BasicPoly &bp)
11{
12 const Ring *K = P->getCoefficients();
13 const Monoid *M = P->getMonoid();
14 int nvars = M->n_vars();
15
16 ring_elem result = P->from_long(0);
17 std::vector<int> exp(nvars, 0);
18 monomial monom = M->make_one();
19
20 int monomStart = 0;
21 for (size_t i = 0; i < bp.mCoefficients.size(); ++i)
22 {
23 // Convert coefficient: mpz_class -> ring_elem in coefficient ring
24 ring_elem coeff = K->from_int(bp.mCoefficients[i].get_mpz_t());
25
26 // Convert monomial: varpower format -> exponent vector -> encoded monomial
27 int monomLen = bp.mMonomials[monomStart];
28 std::fill(exp.begin(), exp.end(), 0);
29 for (int j = monomStart + 1; j < monomStart + monomLen; j += 2)
30 {
31 int var = bp.mMonomials[j];
32 int e = bp.mMonomials[j + 1];
33 if (var >= 0 && var < nvars)
34 exp[var] = e;
35 }
36 M->from_expvector(exp.data(), monom);
37
38 // Create the term and add it to the result
39 ring_elem term = P->make_flat_term(coeff, monom);
40 result = P->add(result, term);
41
42 monomStart += monomLen;
43 }
44
45 M->remove(monom);
46 return result;
47}

49RingElem RingElem::fromString(const Ring *R, const std::string &s)
50{
51 // Polynomial rings (including WeylAlgebra, skew, quotient rings, etc.)
52 const PolynomialRing *P = R->cast_to_PolynomialRing();
53 if (P != nullptr)
54 {
55 const Monoid *M = P->getMonoid();
56 std::vector<std::string> varnames = M->variableNames();
57 BasicPoly bp = parseBasicPoly(s, varnames);
58 ring_elem result = basicPolyToRingElem(P, bp);
59 return RingElem(R, result);
60 }
61
62 // Base rings: parse as an integer
63 // Works for ZZ, ZZ/p, QQ (integers only), and other rings with from_int
64 mpz_class val(s);
65 return RingElem(R, R->from_int(val.get_mpz_t()));
66}
