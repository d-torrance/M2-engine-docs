40MonomialOrdering* offset(const MonomialOrdering* mo, int amount)
41{
42 auto result = makeOrdering(mo->len);
43 for (int i = 0; i < mo->len; ++i)
44 {
45 auto part = mo->array[i];
46 if (part->type != MO_WEIGHTS)
47 result->array[i] = makePart(part->type, part->nvars, part->wts);
48 else
49 {
50 auto shifted = makePart(MO_WEIGHTS, amount + part->nvars, nullptr);
51 shifted->wts = getmemvectortype(int, shifted->nvars);
52 for (int j = 0; j < amount; ++j) shifted->wts[j] = 0;
53 for (int j = amount; j < shifted->nvars; ++j)
54 shifted->wts[j] = part->wts[j - amount];
55 result->array[i] = shifted;
56 }
57 }
58 return result;
59}

143MonomialOrdering* MonomialOrderings::GRevLex(const std::vector<int>& degrees, int packing)
144{
145 bool allOne = true;
146 for (auto degree : degrees)
147 if (degree <= 0)
148 {
149 ERROR("grevlex: expected all degrees to be positive");
150 return nullptr;
151 }
152 else if (degree > 1)
153 allOne = false;
154
155 enum MonomialOrdering_type type;
156 if (allOne)
157 {
158 if (packing == 2)
159 type = MO_GREVLEX2;
160 else if (packing == 4)
161 type = MO_GREVLEX4;
162 else
163 type = MO_GREVLEX;
164 }
165 else
166 {
167 if (packing == 2)
168 type = MO_GREVLEX2_WTS;
169 else if (packing == 4)
170 type = MO_GREVLEX4_WTS;
171 else
172 type = MO_GREVLEX_WTS;
173 }
174 auto result = makeOrdering(1);
175 result->array[0] = makePart(type, degrees.size(), allOne ? nullptr : degrees.data());
176 return result;
177}

179MonomialOrdering* MonomialOrderings::join(const std::vector<MonomialOrdering*>& orderings)
180{
181 int length = 0;
182 for (auto ordering : orderings)
183 for (int j = 0; j < ordering->len; ++j)
184 if (isGood(ordering->array[j])) ++length;
185
186 auto result = makeOrdering(length);
187 int next = 0;
188 int variablesSoFar = 0;
189 for (auto ordering : orderings)
190 for (int j = 0; j < ordering->len; ++j)
191 {
192 auto part = ordering->array[j];
193 if (!isGood(part)) continue;
194 if (part->type != MO_WEIGHTS)
195 variablesSoFar += part->nvars;
196 else
197 {
198 auto shifted = makePart(MO_WEIGHTS, variablesSoFar + part->nvars, nullptr);
199 shifted->wts = getmemvectortype(int, shifted->nvars);
200 for (int k = 0; k < variablesSoFar; ++k) shifted->wts[k] = 0;
201 for (int k = variablesSoFar; k < shifted->nvars; ++k)
202 shifted->wts[k] = part->wts[k - variablesSoFar];
203 part = shifted;
204 }
205 result->array[next++] = part;
206 }
207 return result;
208}

210MonomialOrdering* MonomialOrderings::product(const std::vector<MonomialOrdering*>& orderings)
211{
212 int length = 0;
213 for (auto ordering : orderings) length += ordering->len;
214 auto result = makeOrdering(length);
215 int next = 0;
216 int amount = 0;
217 for (auto ordering : orderings)
218 {
219 auto shifted = offset(ordering, amount);
220 for (int j = 0; j < shifted->len; ++j) result->array[next++] = shifted->array[j];
221 amount += numberOfVariables(ordering);
222 }
223 return result;
224}

262bool MonomialOrderings::isLex(const MonomialOrdering* mo)
263{
264 // The monomial order is lex if what?
265 // one lex block, no grevlex blocks, no weightvector blocks.
266 // only: lex block and position blocks are allowed.
267 int blocks = 0;
268 for (int i = 0; i < mo->len; ++i)
269 switch (mo->array[i]->type)
270 {
271 case MO_LEX: case MO_LEX2: case MO_LEX4: ++blocks; break;
272 case MO_POSITION_UP: case MO_POSITION_DOWN: break;
273 default: return false;
274 }
275 return blocks == 1;
276}

320std::vector<int> MonomialOrderings::nonTermOrderVariables(const MonomialOrdering* mo)
321{
322 auto relation = std::vector<int>(numberOfVariables(mo), 0);
323 int next = 0;
324 for (int i = 0; i < mo->len; ++i)
325 {
326 auto part = mo->array[i];
327 switch (part->type)
328 {
329 case MO_LEX: case MO_LEX2: case MO_LEX4: case MO_GREVLEX:
330 case MO_GREVLEX2: case MO_GREVLEX4: case MO_GREVLEX_WTS:
331 case MO_GREVLEX2_WTS: case MO_GREVLEX4_WTS: case MO_LAURENT:
332 case MO_NC_LEX:
333 for (int j = 0; j < part->nvars; ++j, ++next)
334 if (relation[next] == 0) relation[next] = 1;
335 break;
336 case MO_LAURENT_REVLEX: case MO_REVLEX:
337 for (int j = 0; j < part->nvars; ++j, ++next)
338 if (relation[next] == 0) relation[next] = -1;
339 break;
340 case MO_WEIGHTS:
341 {
342 int top = std::min(part->nvars, static_cast<int>(relation.size()));
343 for (int j = next; j < top; ++j)
344 if (relation[j] == 0)
345 {
346 if (part->wts[j] > 0)
347 relation[j] = 1;
348 else if (part->wts[j] < 0)
349 relation[j] = -1;
350 }
351 }
352 break;
353 case MO_POSITION_UP: case MO_POSITION_DOWN: break;
354 }
355 }
356 std::vector<int> result;
357 for (int i = 0; i < relation.size(); ++i)
358 {
359 if (relation[i] == 0) INTERNAL_ERROR("relation[i] should not be 0");
360 if (relation[i] < 0) result.push_back(i);
361 }
362 return result;
363}

388bool monomialOrderingToMatrix(const MonomialOrdering& mo,
389 std::vector<int>& matrix,
390 bool& baseIsRevLex,
391 int& componentDirection,
392 int& componentIsBeforeRow)
393{
394 int nvars = MonomialOrderings::numberOfVariables(&mo);
395 baseIsRevLex = true;
396 enum LastBlock { LEX, REVLEX, WEIGHTS, NONE };
397 LastBlock last = NONE;
398 int rows = 0;
399 int firstvar = 0;
400 componentDirection = 0;
401 componentIsBeforeRow = -2;
402 size_t lastElement = 0;
403 for (int i = 0; i < mo.len; ++i)
404 {
405 mon_part part = mo.array[i];
406 switch (part->type)
407 {
408 case MO_LEX: case MO_LEX2: case MO_LEX4:
409 lastElement = matrix.size();
410 for (int j = 0; j < part->nvars; ++j) writeRow(matrix, nvars, firstvar + j, 1);
411 last = LEX;
412 firstvar += part->nvars;
413 rows += part->nvars;
414 break;
415 case MO_GREVLEX: case MO_GREVLEX2: case MO_GREVLEX4:
416 case MO_GREVLEX_WTS: case MO_GREVLEX2_WTS: case MO_GREVLEX4_WTS:
417 writeWeights(matrix, nvars, firstvar, part->wts, part->nvars);
418 lastElement = matrix.size();
419 for (int j = part->nvars - 1; j >= 1; --j) writeRow(matrix, nvars, firstvar + j, -1);
420 last = REVLEX;
421 firstvar += part->nvars;
422 rows += part->nvars;
423 break;
424 case MO_REVLEX:
425 lastElement = matrix.size();
426 for (int j = part->nvars - 1; j >= 0; --j) writeRow(matrix, nvars, firstvar + j, -1);
427 last = REVLEX;
428 firstvar += part->nvars;
429 rows += part->nvars;
430 break;
431 case MO_WEIGHTS:
432 writeWeights(matrix, nvars, 0, part->wts, part->nvars > nvars ? nvars : part->nvars);
433 ++rows;
434 lastElement = matrix.size();
435 last = WEIGHTS;
436 break;
437 case MO_LAURENT: case MO_LAURENT_REVLEX: case MO_NC_LEX:
438 return false;
439 case MO_POSITION_UP:
440 componentDirection = 1;
441 componentIsBeforeRow = rows;
442 break;
443 case MO_POSITION_DOWN:
444 componentDirection = -1;
445 componentIsBeforeRow = rows;
446 break;
447 default: break;
448 }
449 }
450 if (last == LEX)
451 {
452 // last block was lex, so use lex tie-breaker
453 matrix.resize(lastElement);
454 if (rows == componentIsBeforeRow) componentIsBeforeRow = -1;
455 baseIsRevLex = false;
456 }
457 else if (last == REVLEX)
458 {
459 // last block was revlex, so use revlex tie-breaker
460 if (rows == componentIsBeforeRow) componentIsBeforeRow = -1;
461 matrix.resize(lastElement);
462 }
463 // last block is a weight vector, so use revlex as the tie-breaker.
464 // nothing to change here.
465 return true;
466}
