100void ReducedGB_Field::remainder(POLY &f, bool use_denom, ring_elem &denom)
101{
102 gbvector head;
103 gbvector *frem = &head;
104 frem->next = nullptr;
105 POLY h = f;
106 exponents_t EXP = ALLOCATE_EXPONENTS(R->exponent_byte_size());
107
108 while (!R->gbvector_is_zero(h.f))
109 {
110 R->gbvector_get_lead_exponents(F, h.f, EXP);
111 int x = h.f->comp;
112 Bag *b;
113 if (Rideal != nullptr && Rideal->search_expvector(EXP, b))
114 {
115 const gbvector *g = originalR->quotient_gbvector(b->basis_elem());
116 R->gbvector_reduce_lead_term(
117 F, Fsyz, head.next, h.f, h.fsyz, g, nullptr, use_denom, denom);
118 }
119 else
120 {
121 int w = T->find_divisor(EXP, x);
122 if (w >= 0)
123 {
124 POLY g = polys[w];
125 R->gbvector_reduce_lead_term(F,
126 Fsyz,
127 head.next,
128 h.f,
129 h.fsyz,
130 g.f,
131 g.fsyz,
132 use_denom,
133 denom);
134 }
135 else
136 {
137 frem->next = h.f;
138 frem = frem->next;
139 h.f = h.f->next;
140 frem->next = nullptr;
141 }
142 }
143 }
144
145 h.f = head.next;
146
147 ring_elem denom1;
148 ring_elem one_elem = R->get_flattened_coefficients()->one();
149 denom1 = R->get_flattened_coefficients()->one(); // is replaced on next line
150 originalR->get_quotient_info()->gbvector_normal_form(
151 Fsyz, h.fsyz, true, denom1);
152 if (EQ != R->get_flattened_coefficients()->compare_elems(denom1, one_elem))
153 {
154 // multiply h.f by denom1.
155 R->gbvector_mult_by_coeff_to(h.f, denom1);
156 }
157 if (use_denom)
158 {
159 R->get_flattened_coefficients()->mult_to(denom, denom1);
160 }
161
162 f.f = h.f;
163 f.fsyz = h.fsyz;
164}
