3// RingElem: A lightweight value-semantics wrapper around (const Ring*, ring_elem).
4// Unlike RingElement (which is heap-allocated and returns pointers from operators),
5// RingElem lives on the stack and operators return values, making test code concise:
6//
7// auto x = RingElem::var(W, 0);
8// auto Dx = RingElem::var(W, 2);
9// auto one = RingElem::fromInt(W, 1);
10// EXPECT_EQ(Dx * x - x * Dx, one);
11//
12// auto f = RingElem::fromString(R, "x^2+3*x*y-1");
13

33 public:
34 // Primary constructor: wraps a pre-computed ring_elem.
35 // Use the static factories below for common cases.
36 RingElem(const Ring *R, ring_elem f) : mRing(R), mValue(f) {}
37
38 // Factory: create the i-th variable of the ring
39 static RingElem var(const Ring *R, int i) { return RingElem(R, R->var(i)); }
40
41 // Factory: create a ring element from an integer
42 static RingElem fromInt(const Ring *R, long n)
43 {
44 return RingElem(R, R->from_long(n));
45 }
46
47 // Factory: create a ring element from a string.
48 // For polynomial rings: parses "x^2+3*x*y-1" using variable names from the ring.
49 // For base rings (ZZ, ZZ/p): parses an integer.
50 // Throws parsing_error on failure.
51 // NOTE: fromString and toString are not yet inverses of each other.
52 // toString outputs e.g. "x3+2xyz" while fromString expects "x^3+2*x*y*z".
53 // TODO: make these round-trip compatible.
54 // TODO: fromDouble
55 static RingElem fromString(const Ring *R, const std::string &s);
