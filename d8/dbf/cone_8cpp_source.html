32const Matrix /* or null */ *rawFourierMotzkin(const Matrix *C)
33{
34 try
35 {
36 // TODO: generalize the input type, in particular to allow lineality space
37 const Ring *R = C->get_ring();
38 const size_t c = C->n_cols(); // rank of ambient lattice
39 const size_t r = C->n_rows(); // number of cone inequalities
40
41 auto ineqs = libnormaliz::Matrix<Integer>(r, c);
42 for (size_t i = 0; i < r; i++)
43 for (size_t j = 0; j < c; j++)
44 // libnormaliz uses A*x >= 0, Macaulay2 uses A*x <= 0
45 ineqs[i][j] = (-1) * static_cast<Integer>(C->elem(i, j).get_mpz());
46
47 auto cone = libnormaliz::Cone<Integer>(libnormaliz::Type::inequalities, ineqs);
48 auto rays = cone.getExtremeRays();
49 size_t n = rays.size(); // number of extremal rays
50
51 MatrixConstructor mat(R->make_FreeModule(n), c);
52 for (size_t i = 0; i < n; i++)
53 for (size_t j = 0; j < c; j++)
54 {
55 mpz_ptr z = newitem(__mpz_struct);
56 mpz_init_set(z, rays[i][j].get_mpz_t());
57 mpz_reallocate_limbs(z);
58 mat.set_entry(i, j, ring_elem(z));
59 }
60
61 return mat.to_matrix();
62 } catch (const exc::engine_error &e)
63 {
64 ERROR(e.what());
65 return nullptr;
66 }

125auto decodeArrayArrayInt(M2_arrayint a) -> std::vector<std::vector<int>>
126{
127 // throw an error if the
128 auto nelems = a->array[0];
129 std::vector<std::vector<int>> result;
130 int next = 1;
131 for (int i = 0; i < nelems; ++i)
132 {
133 result.emplace_back();
134 int len = a->array[next];
135 for (int j = 1; j <= len; ++j) result[i].push_back(a->array[next + j]);
136 next += len + 1;
137 }
138 return result;
139}

196// Enumerate points in the polyhedron `Ax <= b`., s.t. each entry has absolute
197// value <= B. max_N_out can be set to obtain a subset of all the points.
198// max_N_nodes is an internal counter that must be set high enough. If we hit
199// that bound, we throw an error.
200//
201// The underlying box_enum function
202// works with Hx >= b, so we negate A and b on the way in.
203MutableMatrix *rawLatticePoints(const Matrix *A,
204 const Matrix *b,
205 int B,
206 long max_N_out,
207 long max_N_nodes)
208{
209 try
210 {
211 const size_t n_hyps = A->n_rows();
212 const size_t dim = A->n_cols();
213
214 if (b->get_ring() != globalZZ)
215 {
216 ERROR("rawLatticePoints: b must be a matrix over ZZ");
217 return nullptr;
218 }
219 if (static_cast<size_t>(b->n_rows()) != n_hyps || b->n_cols() != 1)
220 {
221 ERROR("rawLatticePoints: b must be a column matrix with n_rows(A) rows");
222 return nullptr;
223 }
224
225 // Marshal A entries to vector<vector<int>>, negating to convert
226 // A*x <= b into (-A)*x >= -b for the box_enum convention.
227 std::vector<std::vector<int>> Hvec(n_hyps, std::vector<int>(dim));
228 for (size_t i = 0; i < n_hyps; i++)
229 for (size_t j = 0; j < dim; j++)
230 {
231 mpz_srcptr z = A->elem(i, j).get_mpz();
232 if (mpz_fits_sint_p(z) == 0)
233 {
234 ERROR("rawLatticePoints: entry of A does not fit in a C int");
235 return nullptr;
236 }
237 Hvec[i][j] = -static_cast<int>(mpz_get_si(z));
238 }
239
240 std::vector<int> rhsvec(n_hyps);
241 for (size_t i = 0; i < n_hyps; i++)
242 {
243 mpz_srcptr z = b->elem(i, 0).get_mpz();
244 if (mpz_fits_sint_p(z) == 0)
245 {
246 ERROR("rawLatticePoints: entry of b does not fit in a C int");
247 return nullptr;
248 }
249 rhsvec[i] = -static_cast<int>(mpz_get_si(z));
250 }
251
252 auto result = M2::cytools::latticePoints(
253 static_cast<int>(dim), B, Hvec, rhsvec, max_N_out, max_N_nodes);
254
255 // Output: rows = ambient coords (dim), cols = lattice points.
256 const size_t n_points = result.points.size();
257 MutableMatrix *M =
258 MutableMatrix::zero_matrix(globalZZ, dim, n_points, /*dense*/ true);
259 for (size_t i = 0; i < n_points; i++)
260 for (size_t j = 0; j < dim; j++)
261 M->set_entry(j, i, globalZZ->from_long(result.points[i][j]));
262 return M;
263 } catch (const std::runtime_error &e)
264 {
265 // catches both std::runtime_error from latticePoints() and
266 // exc::engine_error (which derives from std::runtime_error)
267 ERROR(e.what());
268 return nullptr;
269 }
270}

328// This function attempts to find an interior point of the cone `Ax <= 0`.
329// (where A is a matrix over ZZ). It either finds such a point (if the cone is
330// full dimensional) or if not, it gives non-negative weights certifying that
331// the cone is not full dimensional.
332//
333// Return value is a 1-row MutableMatrix over RR(53):
334// if full-dimensional: 2 + n columns, [1.0, tStar, interior point]
335// else : 2 + m columns, [0.0, tStar, dual certificate]
336// I am a little unclear yet when this will give incorrect answers.
337// Returns nullptr and sets ERROR, if an
338MutableMatrix /* or null */ *rawConeInteriorPoint(const Matrix *A)
339{
340 try
341 {
342 const size_t m = A->n_rows(); // number of inequalities
343 const size_t n = A->n_cols(); // ambient dimension
344
345 std::vector<int> Avec(m * n);
346 for (size_t i = 0; i < m; i++)
347 for (size_t j = 0; j < n; j++)
348 {
349 mpz_srcptr z = A->elem(i, j).get_mpz();
350 if (mpz_fits_sint_p(z) == 0)
351 {
352 ERROR("rawConeInteriorPoint: entry of A does not fit in a C int");
353 return nullptr;
354 }
355 Avec[i * n + j] = -static_cast<int>(mpz_get_si(z));
356 }
357
358 ConeResult cr = coneInteriorPoint(static_cast<int>(m),
359 static_cast<int>(n),
360 Avec);
361
362 const Ring *RR = IM2_Ring_RRR(53);
363 const std::vector<double> &tail =
364 cr.fullDimensional ? cr.interiorPoint : cr.dualCert;
365 MutableMatrix *M = MutableMatrix::zero_matrix(
366 RR, 1, 2 + tail.size(), /*dense*/ true);
367
368 ring_elem r;
369 RR->from_double(cr.fullDimensional ? 1.0 : 0.0, r);
370 M->set_entry(0, 0, r);
371 RR->from_double(cr.tStar, r);
372 M->set_entry(0, 1, r);
373 for (size_t k = 0; k < tail.size(); k++)
374 {
375 RR->from_double(tail[k], r);
376 M->set_entry(0, 2 + k, r);
377 }
378
379 return M;
380 } catch (const exc::engine_error &e)
381 {
382 ERROR(e.what());
383 return nullptr;
384 }
