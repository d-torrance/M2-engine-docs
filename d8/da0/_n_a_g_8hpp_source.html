49// PointArray
50class PointArray
51{
52 public:
53 using RealVector = std::vector<double>;
54 using Weight = double;
55 PointArray(Weight epsilon, const RealVector& weights)
56 : mEpsilon(epsilon), mWeights(weights)
57 {
58 }
59 PointArray(Weight epsilon, int n) : mEpsilon(epsilon)
60 {
61 double s = 0;
62 for (int i = 0; i < n; i++)
63 {
64 double r = rand();
65 s += r;
66 mWeights.push_back(r);
67 }
68 for (int i = 0; i < n; i++) mWeights[i] /= s;
69 }
70 virtual ~PointArray() {
71 // std::cerr << "entering ~PointArray()" << std::endl;
72 }
73 int lookup_or_append(const RealVector& a)
74 {
75 Weight w = weight(a);
76 int p = lookup(a);
77 if (p != -1) return p;
78 if (mMap.find(w) != mMap.end()) return -1;
79 int ret = static_cast<int>(mPoints.size());
80 mPoints.push_back(a);
81 mMap.insert(std::pair<Weight, int>(w, ret));
82 return ret;
83 }
84 int lookup(const RealVector& a) const
85 {
86 Weight w = weight(a);
87 auto le = left(w);
88 auto ri = right(w);
89 for (auto i = le; i != ri; ++i)
90 if (are_same(mPoints[i->second], a)) return i->second;
91 return -1;
92 }
93 Weight weight(const RealVector& a) const
94 {
95 Weight w = 0;
96 auto n = mWeights.size();
97 assert(n == a.size());
98 for (int i = 0; i < n; ++i) w += mWeights[i] * a[i];
99 return w;
100 }
101 bool are_same(const RealVector& a, const RealVector& b) const
102 {
103 auto n = a.size();
104 assert(n == b.size());
105 for (int i = 0; i < n; ++i)
106 if (fabs(a[i] - b[i]) >= mEpsilon) return false;
107 return true;
108 }
109 void text_out(buffer& o) const
110 {
111 o << "PointArray( " << mPoints.size() << " points: ";
112 for (auto i = mMap.begin(); i != mMap.end(); ++i) o << i->second << " ";
113 o << ")" << newline;
114 }
115
116 private:
117 std::map<Weight, int> mMap;
118 std::vector<RealVector> mPoints;
119 Weight mEpsilon;
120 RealVector mWeights;
122 decltype(mMap)::const_iterator left(Weight key) const
123 {
124 return mMap.lower_bound(key - mEpsilon);
125 }
126 decltype(mMap)::const_iterator right(Weight key) const
127 {
128 return mMap.upper_bound(key + mEpsilon);
129 }
130};

277inline complex complex::operator*(complex c)
278{
279 complex tmp;
280 tmp.real = (real * c.real) - (imag * c.imag);
281 tmp.imag = (real * c.imag) + (imag * c.real);
282 return tmp;
283}
284
285inline complex complex::operator/(complex c)
286{
287 double div = (c.real * c.real) + (c.imag * c.imag);
288 complex tmp;
289 tmp.real = (real * c.real) + (imag * c.imag);
290 tmp.real /= div;
291 tmp.imag = (imag * c.real) - (real * c.imag);
292 tmp.imag /= div;
293 return tmp;
294}
295
296inline complex complex::getconjugate() const
297{
298 complex tmp;
299 tmp.real = this->real;
300 tmp.imag = this->imag * -1;
301 return tmp;
302}

304inline complex complex::getreciprocal()
305{
306 complex t;
307 t.real = real;
308 t.imag = imag * -1;
309 double div;
310 div = (real * real) + (imag * imag);
311 t.real /= div;
312 t.imag /= div;
313 return t;
314}
315
316inline double complex::getmodulus()
317{
318 double z;
319 z = (real * real) + (imag * imag);
320 z = sqrt(z);
321 return z;
322}

325inline double complex::getimaginary() const { return imag; }
326inline bool complex::operator==(complex c)
327{

329}

600 void evaluate_slpHxt(int n, const complex* x0t0, complex* Hxt)
601 {
602 if (is_projective)
603 {
604 complex* SxS = newarray_atomic(complex, (n + 1) * (n - 1));
605 complex* TxT = newarray_atomic(complex, (n + 1) * (n - 1));
606 const complex *x0 = x0t0, *t0 = x0t0 + n;
607 const double t = (*(double*)t0) * bigT; // t0 should be real
608 slpSxS->evaluate(n, x0, SxS);
609 slpTxT->evaluate(n, x0, TxT);
610 double c = cos(t), s = sin(t),
611 sqrt_one_minus_productST_2 = sqrt(1 - productST * productST);
612 double mS = c - productST * s / sqrt_one_minus_productST_2;
613 double mT = s / sqrt_one_minus_productST_2;
614 int j, i;
615 for (j = 0; j < n - 1; j++)
616 for (i = 0; i < n; i++)
617 *(Hxt + i * n + j) =
618 *(SxS + i * (n - 1) + j) * mS + *(TxT + i * (n - 1) + j) * mT;
619 j = n - 1; // last column
620 for (i = 0; i < n; i++) *(Hxt + i * n + j) = (x0 + i)->getconjugate();
621 i = n; // last row = H_t
622 mS = -s - productST * c / sqrt_one_minus_productST_2;
623 mT = c / sqrt_one_minus_productST_2;
624 for (j = 0; j < n - 1; j++)
625 {
626 complex tt =
627 *(SxS + i * (n - 1) + j) * mS + *(TxT + i * (n - 1) + j) * mT;
628 *(Hxt + i * n + j) = tt;
629 }
630 *(Hxt + n * n + n - 1) = complex(0.); // last row and column
631 freemem(SxS);
632 freemem(TxT);
633 }
634 else
635 slpHxt->evaluate(n + 1, x0t0, Hxt);
636 }
