81M2_arrayint rawMonomialOrderingToMatrix(const MonomialOrdering* mo)
82{
83 bool base;
84 std::vector<int> matrix;

86 int componentDirection = 0;
87 if (!monomialOrderingToMatrix(*mo, matrix, base, componentDirection, componentIsBeforeRow)) return nullptr;
88 int top = static_cast<int>(matrix.size());
89 M2_arrayint result = M2_makearrayint(top + 3);
90 for (int i = 0; i < top; ++i) result->array[i] = matrix[i];
91 result->array[top] = base ? 1 : 0;
92 result->array[top + 1] = componentDirection;
93 result->array[top + 2] = componentIsBeforeRow;
94 return result;
