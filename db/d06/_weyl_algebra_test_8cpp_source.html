110TEST_F(WeylAlgebraTest, multinomial)
111{
112 // multinomial(c, top, bottom) = c * product_i binomial(top[i], bottom[i])
113 // where the product is over the _nderivatives derivative pairs.
114 // Here _nderivatives = 2 (for Dx, Dy).
115
116 ring_elem one = K->from_long(1);
117
118 // All bottom entries zero: result should be c * 1 = c
119 {
120 int top[] = {5, 3};
121 int bottom[] = {0, 0};
122 expectEqualLong(WeylAlgebraTestAccessor::multinomial(W, one, top, bottom), 1);
123 }
124
125 // Single nonzero bottom entry: should give binomial(5,2) = 10
126 {
127 int top[] = {5, 3};
128 int bottom[] = {2, 0};
129 expectEqualLong(WeylAlgebraTestAccessor::multinomial(W, one, top, bottom), 10);
130 }
131
132 // Both entries nonzero: binomial(5,2) * binomial(3,1) = 10 * 3 = 30
133 {
134 int top[] = {5, 3};
135 int bottom[] = {2, 1};
136 expectEqualLong(WeylAlgebraTestAccessor::multinomial(W, one, top, bottom), 30);
137 }
138
139 // With a scalar coefficient: 7 * binomial(4,2) * binomial(6,3) = 7 * 6 * 20 = 840
140 {
141 ring_elem seven = K->from_long(7);
142 int top[] = {4, 6};
143 int bottom[] = {2, 3};
144 expectEqualLong(WeylAlgebraTestAccessor::multinomial(W, seven, top, bottom), 840);
145 }
146}
